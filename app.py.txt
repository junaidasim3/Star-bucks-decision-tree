
import streamlit as st
import pandas as pd
import joblib

st.set_page_config(
    page_title="Starbucks Smart Recommender",
    page_icon="☕",
    layout="centered"
)

@st.cache_resource
def load_model():
    return joblib.load("starbucks_decision_tree.pkl")

model = load_model()

st.title("☕ Starbucks Smart Drink Recommender")
st.write("Discover a drink based on your preferences!")

strength = st.selectbox(
    "Coffee Strength",
    [1, 2, 3],
    format_func=lambda x: {
        1: "Mild", 2: "Medium", 3: "Strong"
    }[x]
)

sweetness = st.selectbox(
    "Sweetness Level",
    [1, 2, 3],
    format_func=lambda x: {
        1: "Low", 2: "Medium", 3: "High"
    }[x]
)

temperature = st.selectbox(
    "Drink Temperature",
    [0, 1],
    format_func=lambda x: {
        0: "Hot", 1: "Cold"
    }[x]
)

milk = st.selectbox(
    "Milk Preference",
    [1, 2],
    format_func=lambda x: {
        1: "Dairy Milk", 2: "Plant-Based Milk"
    }[x]
)

if st.button("Recommend My Drink"):
    features = pd.DataFrame(
        [[strength, sweetness, temperature, milk]],
        columns=[
            "Coffee_Strength",
            "Sweetness",
            "Temperature",
            "Milk_Preference"
        ]
    )

    drink = model.predict(features)[0]
    st.success(f"Your recommended drink is: {drink}")

    st.caption(
        "Recommendation generated by your trained "
        "Decision Tree model."
    )
